 function module:newStun(parent, duration, name)
	local stun = Instance.new("BoolValue")
	if name then
		stun.Name = name
	else
		stun.Name = "stun"
	end
	stun.Parent = parent
	game.Debris:AddItem(stun, duration)
	return stun
end

function module:flash(char, fill, outline, duration)
	local h = Instance.new("Highlight")
	h.DepthMode = Enum.HighlightDepthMode.Occluded
	h.FillColor = fill
	h.OutlineColor = outline
	h.FillTransparency = 0.5
	h.OutlineTransparency = 0
	h.Parent = char
	
	game.Debris:AddItem(h, duration)
	
	local ti = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
	local t = ts:Create(h, ti, {FillTransparency = 1})
	local t2 = ts:Create(h, ti, {OutlineTransparency = 1})
	t:Play()
	t2:Play()
end

function module:iframes(char, toggle)
	char:SetAttribute("iframes", toggle)
end

function module:removeStuns(char)
	for _,v in char:GetChildren() do
		if v:IsA("BoolValue") and v.Name == "stun" then
			v:Destroy()
		end
	end 
end

function module:hitEffect(victim, name, customParent)
	local vfxDest
	
	if customParent then
		vfxDest = customParent
	elseif victim:FindFirstChild("Torso") then
		vfxDest = victim.Torso
	else
		vfxDest = victim.HumanoidRootPart
	end

	if vfxDest:FindFirstChild(name) then
		if vfxDest:FindFirstChild(name):IsA("Attachment") then
			for _,p in vfxDest:FindFirstChild(name):GetChildren() do
				p:Emit(p:GetAttribute("EmitCount"))
			end
			return
		end
	end

	if HitVFX:FindFirstChild(name) then
		local vfxClone = HitVFX[name]:Clone()
		vfxClone.Parent = vfxDest
		for _,p in vfxClone:GetChildren() do
			p:Emit(p:GetAttribute("EmitCount"))
		end
	else
		warn("invalid vfx attachment: " + name)
	end
end

function module:LevelUpFX(root, tag)
	for _,v in LevelUp:GetChildren() do
		if v.Name:find(tag) then
			local c = v:Clone()
			c.Parent = root
			c:Emit(c:GetAttribute("EmitCount"))
			game.Debris:AddItem(c, 7)
		end
	end
	module:playSound(script:FindFirstChild("LevelUp"), root)
end

function module:playSound(sound, dir, start, duration, speed)
	local newSound = sound:Clone()
	newSound.Parent = dir
	
	if start then
		newSound.TimePosition = start
	end
	if speed then
		newSound.PlaybackSpeed = speed
	end
	
	newSound:Play()
	if duration then
		game.Debris:AddItem(newSound, duration)
	else
		game.Debris:AddItem(newSound, newSound.TimeLength + 1)
	end
end

function module:hit(player, victim, move_data)

	local hitSound = Sounds:FindFirstChild(move_data["vfx"])
	if hitSound then
		module:playSound(hitSound, victim.HumanoidRootPart)
	else
		warn("invalid hit sound: "..move_data["vfx"])
	end

	print(victim)
	
	if move_data["kill"] then
		victim.Humanoid.Health = 0
	else
		local deduction = victim.Humanoid.Health - move_data["damage"]
		if deduction <= 0 then
			-- victim.HumanoidRootPart.Anchored = true
			-- Animator.PlayAnimation(downed_anim, victim)
			
			local isPlayer = false
			
			if game.Players:FindFirstChild(victim.Name) then
				print("is a player")
				isPlayer = true
			else
				print("not a player")
			end

			warn("ragdolled "..victim.Name)
			RagdollModule.Ragdoll(victim)
			victim:SetAttribute("downed", true)
			
			if victim.Torso:FindFirstChild("executePrompt") then
				victim.Torso.executePrompt.Enabled = true
			end
			if victim.Torso:FindFirstChild("DefaultFun") then
				victim.Torso.DefaultFun.Enabled = true
			end
			if victim.Torso:FindFirstChild("carryPrompt") then
				victim.Torso.carryPrompt.Enabled = true
			end
			
			if isPlayer then
				ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt}, false)
			end

			warn("downed "..victim.Name)
			
						
			
			task.spawn(function()
				while victim and victim:GetAttribute("downed") do
					victim.Humanoid.WalkSpeed = 0 
					wait(1)
					victim.Head.stats.Frame.TextLabel.TextColor3 = Color3.fromRGB(255, 42, 66)
					local ti = TweenInfo.new(0.5, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
					local t = ts:Create(victim.Head.stats.Frame.TextLabel, ti, {TextColor3 = Color3.fromRGB(255, 255, 255)})
					t:Play()
					if victim and victim:GetAttribute("canGetUp") and victim:FindFirstChild("Humanoid") then
						victim.Humanoid.WalkSpeed = 16
					end
				end
			end)
		else
			victim.Humanoid:TakeDamage(move_data["damage"])
			warn("took ", move_data["damage"], "damage.")
		end
		if victim:GetAttribute("attacking") == true then
			DamageIndicator:FireAllClients(move_data["damage"], victim, true)
		else
			DamageIndicator:FireAllClients(move_data["damage"], victim)
		end
	end
	
	if move_data["stun_anim"] then
		if rs.Animations.bat:FindFirstChild(move_data["stun_anim"]) then
			animator.PlayAnimation(rs.Animations.bat[move_data["stun_anim"]], victim)
		else
			warn("invalid hit anim: "..move_data["stun_anim"])
		end
	end
	
	if victim:GetAttribute("attacking") == true then
		warn("COUNTERED")
		local counterClone = CounterVFX:Clone()
		counterClone.Parent = game.Workspace
		local lookAt = Vector3.new(victim.HumanoidRootPart.Position.X, player.Character.HumanoidRootPart.Position.Y, victim.HumanoidRootPart.Position.Z)
		counterClone.CFrame = CFrame.lookAt(player.Character.HumanoidRootPart.Position, lookAt)
		
		for _,p in counterClone.at:GetChildren() do
			p:Emit(p:GetAttribute("EmitCount"))
		end
		
		module:playSound(Sounds:FindFirstChild("Counter"), victim.HumanoidRootPart)
		
		game.Debris:AddItem(counterClone, 5)
	end
	
	module:newStun(victim, move_data["stun"])
	module:flash(victim, move_data["fill_color"], move_data["outline_color"], 0.6)

	if not victim:GetAttribute("downed") and move_data["ragdoll"] then
		RagdollModule.Ragdoll(victim)
		task.spawn(function()
			wait(move_data["ragdoll"])
			if victim then
				warn("unragdolled "..victim.Name)
				RagdollModule.UnRagdoll(victim)
			end
		end)
	end
	
	module:hitEffect(victim, move_data["vfx"])

	
	local KnockBack = Instance.new("BodyVelocity")
	KnockBack.Parent = victim.HumanoidRootPart--part is the target of the knockback/ the opponent
	KnockBack.MaxForce = Vector3.new(move_data["kb_force"],move_data["kb_force"],move_data["kb_force"])
	KnockBack.Velocity = player.Character.PrimaryPart.CFrame.LookVector * move_data["kb_speed"] -- based on the direction YOUR character is facing
	game.Debris:AddItem(KnockBack, move_data["kb_duration"])
end

function module:setupInteraction(attacker, victim, offset, tag)
	warn("new interaction: "..attacker.Name.." and "..victim.Name)

	RagdollModule.UnRagdoll(victim)
	victim:SetAttribute("canGetUp", false)
	
	ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), false)
	if game:GetService("Players"):GetPlayerFromCharacter(victim) then
		ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(victim), false)
	end

	local attackerRoot = attacker.HumanoidRootPart
	local victimRoot = victim.HumanoidRootPart
	attackerRoot.Anchored = true
	if offset then
		victimRoot.CFrame = attackerRoot.CFrame * offset
	else
		victimRoot.CFrame = attackerRoot.CFrame
	end
	victimRoot.Anchored = true

	module:prompt_changer(false, false, false, false, false, victim, attacker)

	if tag then
		if victim.Torso:FindFirstChild(tag) then
			victim.Torso[tag].Enabled = false
		end
	end

	module:newStun(attacker, math.huge)
	module:newStun(victim, math.huge)
	module:iframes(attacker, true)
	module:iframes(victim, true)
end

local function getMostNuts()
	local plr
	local nutCount = 0
	for _,v in game.Players:GetPlayers() do
		if v:FindFirstChild("leaderstats") then
			if v.leaderstats.Nuts.Value > nutCount then
				nutCount = v.leaderstats.Nuts.Value
				plr = v
			end
		end
	end
	if plr == nil then
		return nil
	end
	return {plr, nutCount}
end

function module:updateMostKills()
	local mostKillsInfo = getMostKills()
	
	if mostKillsInfo == nil then return end

	for _,v in game.Players:GetPlayers() do
		if v.Character and v ~= mostKillsInfo[1] then
			if v.Character.Head:FindFirstChild("society") then
				-- fire event here
				v.Character.Head.society:Destroy()
			end
		end
	end
	
	warn("new max kills: "..mostKillsInfo[1].Name)
	mostKills.Value = mostKillsInfo[1]
end

local awakenInitial = TweenInfo.new(0.4, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
local awakenOutro = TweenInfo.new(1, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)

function module:awaken(char)
	char.Humanoid.AutoRotate = false
	char:SetAttribute("awakened", true)
	module:newStun(char, math.huge)
	module:iframes(char, true)
	animator.PlayAnimation(Animations.other:FindFirstChild("awaken"), char)
	module:playSound(Sounds:FindFirstChild("Awaken"), char.HumanoidRootPart)
	
	Subtitles:FireClient(game.Players:GetPlayerFromCharacter(char), "I'm fired up now!", 1.2)
	
	local circles = VFX.circles:Clone()
	circles.Parent = char.Torso
	
	local par = {}
	local high
	
	for _,v in char:GetChildren() do
		if v:IsA("BasePart") and v.Name ~= "HumanoidRootPart" and v.Name ~= "Torso" then
			for _,b in rs.VFX.Killstreak:GetChildren() do
				local c = b:Clone()
				if c:IsA("ParticleEmitter") then
					c.Parent = v
					table.insert(par, c)
				elseif c:IsA("Highlight") then
					c.Parent = char
					high = c
				end
			end
		end
	end
	
	local t = ts:Create(high, awakenInitial, {OutlineTransparency = 0})
	local t2 = ts:Create(high, awakenInitial, {FillTransparency = 0.5})
	t:Play()
	t2:Play()
	
	AwakenScreen:FireClient(game.Players:GetPlayerFromCharacter(char))
	
	wait(0.4)
	
	circles:Emit(35)
	module:playSound(Sounds:FindFirstChild("Awaken2"), char.HumanoidRootPart)
	
	for _,v in par do
		v.Enabled = true
	end
	
	local t3 = ts:Create(high, awakenOutro, {OutlineTransparency = 1})
	local t4 = ts:Create(high, awakenOutro, {FillTransparency = 1})
	t3:Play()
	t4:Play()
	
	CamShake:FireClient(game.Players:GetPlayerFromCharacter(char), "Explosion")
	
	wait(2)
	
	module:removeStuns(char)
	module:iframes(char, false)
	char.Humanoid.AutoRotate = true
end

local function playSubtitles(attacker, victim, text, dur)

	if game:GetService("Players"):GetPlayerFromCharacter(victim) then
		Subtitles:FireClient(game.Players:GetPlayerFromCharacter(victim), text, dur)
	end
end
function module:grip(attacker, victim, gripType)
	module:setupInteraction(attacker, victim)
	
	local attackerRoot = attacker.HumanoidRootPart
	local victimRoot = victim.HumanoidRootPart
	local victimIsPlayer = game:GetService("Players"):GetPlayerFromCharacter(victim)
	
	if gripType == "Default" then
		local victimAnim = animator.PlayAnimation(gripVictim, victim)
		animator.ChangeAnimationSpeed(victimAnim, 250)
		local attackAnim = animator.PlayAnimation(gripAttacker, attacker)
		animator.ChangeAnimationSpeed(attackAnim, 250)

		wait(0.6)

		module:hit(game.Players:GetPlayerFromCharacter(attacker), victim, MoveData.get_data("GRIP_HIT"))
		DamageIndicator:FireAllClients(MoveData.get_data("GRIP_HIT")["damage"], victim)

		wait(1.5)
		
		module:playSound(Sounds:FindFirstChild("Voice1"), attackerRoot)
		playSubtitles(attacker, victim, "I'll kill you!", 1)
		
		wait(1.4)

		module:hit(game.Players:GetPlayerFromCharacter(attacker), victim, MoveData.get_data("GRIP_FINISHER"))
		DamageIndicator:FireAllClients(MoveData.get_data("GRIP_FINISHER")["damage"], victim)
		if victimIsPlayer then
			RagdollForce:FireClient(game.Players:GetPlayerFromCharacter(victim))
		end

		wait(5)
	elseif gripType == "Finishing Hold" then
		local victimAnim = animator.PlayAnimation(finishHoldVictim, victim)
		animator.ChangeAnimationSpeed(victimAnim, 250)
		local attackAnim = animator.PlayAnimation(finishHoldAttacker, attacker)
		animator.ChangeAnimationSpeed(attackAnim, 250)
		
		module:playSound(Sounds:FindFirstChild("Voice2"), attackerRoot)
		if victimIsPlayer then
			playSubtitles(attacker, victim, "I'm not done beating you up!", 1.5)
		end
		
		module:hit(game.Players:GetPlayerFromCharacter(attacker), victim, MoveData.get_data("BACK_START"))
		
		wait(1.1)
		
		module:playSound(script:FindFirstChild("DioVoice"))
		module:hit(game.Players:GetPlayerFromCharacter(attacker), victim, MoveData.get_data("BACK_BREAK"))
		DamageIndicator:FireAllClients(MoveData.get_data("BACK_BREAK")["damage"], victim)

		wait(0.6)

		module:hit(game.Players:GetPlayerFromCharacter(attacker), victim, MoveData.get_data("GRIP_FINISHER"))
		DamageIndicator:FireAllClients(MoveData.get_data("GRIP_FINISHER")["damage"], victim)
		if victimIsPlayer then
			RagdollForce:FireClient(game.Players:GetPlayerFromCharacter(victim))
		end

		wait(1.2)
	end
	
	attackerRoot.Anchored = false
	victimRoot.Anchored = false
	
	module:removeStuns(attacker)
	module:removeStuns(victim)
	module:iframes(attacker, false)
	module:iframes(victim, false)
	
	ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), true)
	
	local player = game.Players:GetPlayerFromCharacter(attacker)
	player.leaderstats.Kills.Value += 1
	player.leaderstats.Killstreak.Value += 1
	
	module:LevelUpFX(attacker.HumanoidRootPart, "UpKill")
	module:updateMostKills()
	--module:updateMostKillstreak()
end

function module:prompt_changer(executePrompt, DefaultFun, carryPrompt, dropPrompt, stopPrompt, victim, attacker)
	if victim.Torso:FindFirstChild("executePrompt") then
		victim.Torso.executePrompt.Enabled = executePrompt
	end
	if victim.Torso:FindFirstChild("DefaultFun") then
		victim.Torso.DefaultFun.Enabled = DefaultFun
	end
	if victim.Torso:FindFirstChild("stopPrompt") then
		victim.Torso.stopPrompt.Enabled = stopPrompt
	end
	if victim.Torso:FindFirstChild("carryPrompt") then
		victim.Torso.carryPrompt.Enabled = carryPrompt
	end
	if victim.Torso:FindFirstChild("dropPrompt") then
		victim.Torso.dropPrompt.Enabled = dropPrompt
	end
end

function module:particles(juice, n1, n2)
	if juice then
		if juice.Attachment then
			juice.Attachment.Blood:Emit(n1)
		end
		if juice.ParticleEmitter then
			juice.ParticleEmitter:Emit(n2)
		end
	end
end

local function finish_interaction(attacker, victim, attackAnim, victimAnim, tag, juice)
	
	local stopPrompt = victim.Torso:FindFirstChild("stopPrompt")
	local attackerRoot = attacker.HumanoidRootPart
	local victimRoot = victim.HumanoidRootPart
	local isVictimPlayer = game:GetService("Players"):GetPlayerFromCharacter(victim)
	
	attacker:SetAttribute("doing", false)
	attacker:SetAttribute("funMeter", 0)
	
	if(tag == "FD") then
		attackAnim = animator.PlayAnimation(rs.Animations.female[tag].release, attacker)
		victimAnim = animator.PlayAnimation(rs.Animations.male[tag].release, victim)
		module:playSound(script:FindFirstChild("Release"), attackerRoot, 0, 7.5)
		module:playSound(script:FindFirstChild("Eject"), attackerRoot)
	else
		victimAnim = animator.PlayAnimation(rs.Animations.female[tag].release, victim)
		attackAnim = animator.PlayAnimation(rs.Animations.male[tag].release, attacker)
		module:playSound(script:FindFirstChild("Release"), victimRoot, 0, 7.5)
		module:playSound(script:FindFirstChild("Eject"), victimRoot)
		module:playSound(script:FindFirstChild("Test4"), victimRoot)
	end	
	
	local player = game.Players:GetPlayerFromCharacter(attacker)
	
	player.leaderstats.Nuts.Value += 1
	module:LevelUpFX(attacker.HumanoidRootPart, "UpNut")
	module:updateMostNuts()
	
	
	IncreaseHitCount:FireClient(game.Players:GetPlayerFromCharacter(attacker))
	module:particles(juice, 30, 17)
	
	if tag == "Wall" then
		wait(2)
		IncreaseHitCount:FireClient(game.Players:GetPlayerFromCharacter(attacker))

		module:particles(juice, 30, 17)
		module:playSound(script:FindFirstChild("Eject"), victimRoot)

		wait(2)
		IncreaseHitCount:FireClient(game.Players:GetPlayerFromCharacter(attacker))
		module:particles(juice, 30, 17)
		module:playSound(script:FindFirstChild("Eject"), victimRoot)
		module:playSound(script:FindFirstChild("Test4"), victimRoot)
		wait(1)
	elseif(tag == "FD") then
		wait(2)
		IncreaseHitCount:FireClient(game.Players:GetPlayerFromCharacter(attacker))

		module:particles(juice, 30, 17)
		module:playSound(script:FindFirstChild("Eject"), victimRoot)
		module:playSound(script:FindFirstChild("Test4"), victimRoot)

		wait(1)
	else
		wait(3)

		IncreaseHitCount:FireClient(game.Players:GetPlayerFromCharacter(attacker))

		module:particles(juice, 30, 17)

		module:playSound(script:FindFirstChild("Eject"), victimRoot)
		module:playSound(script:FindFirstChild("Test4"), victimRoot)


		wait(4)
	end
	
	attackerRoot.Anchored = false
	victimRoot.Anchored = false
	
	module:removeStuns(attacker)
	module:removeStuns(victim)
	module:iframes(attacker, false)
	module:iframes(victim, false)
	
	module:hitEffect(attacker, "Hearts")
	module:playSound(script:WaitForChild("EnterFun"), victimRoot)
	
	if(tag == "FD") then
		NewChanger:clearMorph(victim, "pp")
		NewChanger:clearMorph(attacker, "boba")
	else
		NewChanger:clearMorph(victim, "boba")
		NewChanger:clearMorph(attacker, "pp")
	end
	
	if isVictimPlayer then
		RagdollModule.Ragdoll(victim)
		RagdollForce:FireClient(game.Players:GetPlayerFromCharacter(victim))
	end
	
	module:prompt_changer(true, true, true, false, false, victim, attacker)
	
	if victim.Torso:FindFirstChild(tag.."Fun") then
		victim.Torso[tag.."Fun"].Enabled = false
	end
	
	ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), stopPrompt, false)
	
	if isVictimPlayer then
		ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(victim), true)
		if victim.Torso:FindFirstChild(tag) then
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt, victim.Torso[tag]}, false)
		else
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt}, false)	
		end
	end
	
	ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), true)
	victim:SetAttribute("carriedBy", "nill")
	attacker:SetAttribute("carrying", "nill")
	victim:SetAttribute("canGetUp", true)
	
end

local function stop_interaction(attacker, victim, tag, victimAnim, attackAnim)
	local attackerRoot = attacker.HumanoidRootPart
	local victimRoot = victim.HumanoidRootPart
	local isVictimPlayer = game:GetService("Players"):GetPlayerFromCharacter(victim)
	
	attacker:SetAttribute("doing", false)
	attackerRoot.Anchored = false
	victimRoot.Anchored = false
	
	module:removeStuns(attacker)
	module:removeStuns(victim)
	module:iframes(attacker, false)
	module:iframes(victim, false)
	
	animator.StopAnimation(victimAnim)
	animator.StopAnimation(attackAnim)
	
	module:hitEffect(victim, "Hearts")
	module:playSound(script:WaitForChild("EnterFun"), victimRoot)
	
	if(tag == "FD") then
		NewChanger:clearMorph(victim, "pp")
		NewChanger:clearMorph(attacker, "boba")
	else
		NewChanger:clearMorph(victim, "boba")
		NewChanger:clearMorph(attacker, "pp")
	end
	
	RagdollModule.Ragdoll(victim)
	module:prompt_changer(true, true, true, false, false,victim, attacker)
	
	if victim.Torso:FindFirstChild(tag.."Fun") then
		victim.Torso[tag.."Fun"].Enabled = false
	end
	
	ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), true)
	
	if isVictimPlayer then
		ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(victim), true)
		local torso = victim:FindFirstChild("Torso")
		local tagExists = torso:FindFirstChild(tag)
		if tagExists then
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt, victim.Torso[tag]}, false)
		else
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt}, false)	
		end

	end
	
	victim:SetAttribute("carriedBy", "nill")
	attacker:SetAttribute("carrying", "nill")
	victim:SetAttribute("canGetUp", true)
	
	--Bug Fixes for Meter
	if attacker:GetAttribute("funMeter") >= 1 then
		attacker:SetAttribute("funMeter", 0.94) -- Place between 0.90 to 0.96
	end
end

local function doing(attacker, victim, victimRoot, soundDelay, juice)
	wait(soundDelay)
	if attacker:GetAttribute("funMeter") + 0.02 <= 1 then
		attacker:SetAttribute("funMeter", attacker:GetAttribute("funMeter") + 0.02)
	else
		attacker:SetAttribute("funMeter", 1)
	end


	if(game.Players:GetPlayerFromCharacter(attacker)) and attacker ~= nil then -- no need, the first one returns FALSE if its null, still adding it bc why not --o--oky
		IncreaseHitCount:FireClient(game.Players:GetPlayerFromCharacter(attacker))
	end

	local bob = victim:FindFirstChild("boba")
	if bob then
		juice = victim.boba.ref.v:FindFirstChild("juice")
		module:particles(juice, 20, 10)
	end

	module:playSound(script["Slap"..math.random(1,3)], victimRoot)
	module:playSound(script["Test"..math.random(1,3)], victimRoot)
end

function module:fun(attacker, victim, tag, param)
	
	--Variables to be used
	local soundDelay = 0.8
	local juice

	local victimRoot = victim.HumanoidRootPart
	local isVictimPlayer = game:GetService("Players"):GetPlayerFromCharacter(victim)
	local victimAnim
	local attackAnim
	local attackerGender = game:GetService("Players"):GetPlayerFromCharacter(attacker):GetAttribute("Gender")
	
	--Set up the primary attributes that are for BOTH
	victim:SetAttribute("carriedBy", attacker.Name)
	attacker:SetAttribute("carrying", victim.Name)
	attacker:SetAttribute("doing", true)
	module:playSound(script:WaitForChild("EnterFun"), victim.HumanoidRootPart)
	

	--Set Interactions based on Gender
	if(attackerGender == "Female" and tag == "Default") then

			NewChanger:loadMorph(attacker, "boba")

		--[[elseif(tag == "Wall") then

			NewChanger:loadMorph(victim, "boba")
			NewChanger:loadMorph(attacker, "pp")

		end]]
		
		juice = attacker.boba.ref.v:FindFirstChild("juice")
		attackAnim = animator.PlayAnimation(rs.Animations.female[tag]["1"], attacker)
		victimAnim = animator.PlayAnimation(rs.Animations.male[tag]["1"], victim)
	else
		if(tag == "Default") then
		elseif(tag == "Wall") then
			attackerRoot.CFrame = param.CFrame * CFrame.new(0,0,4)
		end
		
		module:setupInteraction(attacker, victim, CFrame.new(0,0,-4))
		NewChanger:loadMorph(victim, "boba")
		NewChanger:loadMorph(attacker, "pp")
		
		juice = victim.boba.ref.v:FindFirstChild("juice")
		
		victimAnim = animator.PlayAnimation(rs.Animations.female[tag]["1"], victim)
		attackAnim = animator.PlayAnimation(rs.Animations.male[tag]["1"], attacker)
		
		
	end
	
	module:hitEffect(victim, "Hearts")
	local stopPrompt = victim.Torso:FindFirstChild("stopPrompt")
	ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), stopPrompt, true)
	
	if victim.Torso:FindFirstChild("dropPrompt") then
		victim.Torso.dropPrompt.Enabled = false
	end
	
	local con
	local con2
	con = ChangeFunSpeed.OnServerEvent:Connect(function(plr, funState)
		if plr ~= game.Players:GetPlayerFromCharacter(attacker) then return end
		if funState == 4 then
			if con then
				con:Disconnect()
			end
			if con2 then
				con2:Disconnect()

			finish_interaction(attacker, victim, attackAnim, victimAnim, tag, juice)

		else
			if(tag == "FD") then
				victimAnim = animator.PlayAnimation(rs.Animations.male[tag][funState], victim)
				attackAnim = animator.PlayAnimation(rs.Animations.female[tag][funState], attacker)

				victimAnim = animator.PlayAnimation(rs.Animations.female[tag][funState], victim)
				attackAnim = animator.PlayAnimation(rs.Animations.male[tag][funState], attacker)
			end
			animator.ChangeAnimationSpeed(victimAnim, 350)
			animator.ChangeAnimationSpeed(attackAnim, 350)

		if player ~= game.Players:GetPlayerFromCharacter(attacker) then return end
		local victim = prompt.Parent.Parent
		local attacker = player.Character

		if prompt.Name == "stopPrompt" and prompt.Parent.Parent == victim and player.Character:GetAttribute("downed") == false then
			prompt.Enabled = false

			if con then

			end
			if con2 then
				con2:Disconnect()
			end

			stop_interaction(attacker, victim, tag, victimAnim, attackAnim)
			
		end
	end)

	while (attacker:GetAttribute("doing") == true) do
		doing(attacker, victim, victimRoot, soundDelay, juice)
	end
	
	
	
end

function module:pull(part, c, d, force, response, name, origin)
	if part:FindFirstChild("pull_part") then
		part.pull_part:Destroy()
	end
	local pull_part = Instance.new("Part")
	local at0 = Instance.new("Attachment")
	local at1 = Instance.new("Attachment")
	local align = Instance.new("AlignPosition")
	align.MaxVelocity = math.huge
	align.MaxForce = force
	align.Responsiveness = response

	pull_part.Transparency = 1
	pull_part.Size = part.Size
	pull_part.Anchored = true
	pull_part.CanCollide = false 

	if name then
		pull_part.Name = name
	else
		pull_part.Name = "pull_part"
	end

	pull_part.Massless = true
	pull_part.CanQuery = false
	pull_part.Parent = part

	if typeof(c) == "CFrame" then
		if origin then
			pull_part.CFrame = origin.CFrame * c
		else
			pull_part.CFrame = part.CFrame * c
		end
	elseif typeof(c) == "Vector3" then
		pull_part.Position = c
	end


	at1.Parent = pull_part
	at0.Parent = part

	align.Parent = pull_part
	align.Attachment0 = at0
	align.Attachment1 = at1

	game.Debris:AddItem(pull_part, d)
	game.Debris:AddItem(at1, d)
	game.Debris:AddItem(at0, d)

	return pull_part
end

-- Wall prompts

local allWallPrompts = {}
local currentMapName = "none"

function module:wall_table_append()
	for i,v in pairs(game.Workspace:GetChildren()) do
		if v:FindFirstChild("IsMap") then
			if(v.Name ~= currentMapName) then
				currentMapName = v.Name
				table.clear(allWallPrompts)
				for _,v in v:WaitForChild("WallPrompts"):GetChildren() do
					table.insert(allWallPrompts, v:FindFirstChild("Wall"))
				end
			end
		end
	end
end

function module:carry(attacker, victim)
	if not attacker:FindFirstChild("HumanoidRootPart") or not victim:FindFirstChild("HumanoidRootPart") then return end
	
	local startTime = tick() -- Record the starting time
	warn(attacker.Name.. " picked up "..victim.Name)
	
	
	
	module:wall_table_append()
	
	local attackerRoot = attacker.Torso
	local victimRoot = victim.HumanoidRootPart
	local isVictimPlayer = game:GetService("Players"):GetPlayerFromCharacter(victim)
	
	attackerRoot.Anchored = false
	victimRoot.Anchored = false
	
	module:vic_collision(victim, false)
	
	RagdollModule.UnRagdoll(victim)
	local weld = script.carryWeld:Clone()
	weld.Part0 = attackerRoot
	weld.Part1 = victimRoot
	weld.Parent = attackerRoot
	weld.Enabled = true
	
	victim:SetAttribute("canGetUp", false)
	victim:SetAttribute("downed", true)
	victim:SetAttribute("carriedBy", attacker.Name)
	attacker:SetAttribute("carrying", victim.Name)
	
	
	ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), false)
	if isVictimPlayer then
		ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(victim), false)
	end
	
	module:newStun(attacker, math.huge, "carrying")
	
	for _,v in victim:GetDescendants() do 
		if v:IsA("BasePart") then
			v.Massless = true
		end
	end
	
	module:iframes(victim, true)
	
	local victimAnim = animator.PlayAnimation(carryGrabbed, victim)
	local attackerAnim = animator.PlayAnimation(carryGrabber, attacker)
	
	local dropPrompt = victim.Torso:FindFirstChild("dropPrompt")
	local finishHoldPrompt = victim.Torso:FindFirstChild("finishHoldPrompt")
	
	ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), {dropPrompt, finishHoldPrompt}, true)
	ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), allWallPrompts, true)
	
	module:prompt_changer(false, false, false, false, false,victim, attacker)
	
	module:playSound(script:WaitForChild("Carry"), victimRoot)
	
	local con
	local con2

	con = pps.PromptTriggered:Connect(function(prompt, player)
		if prompt.Name == "dropPrompt" and prompt.Parent.Parent == victim and player.Character:GetAttribute("downed") == false then
			module:vic_collision(victim, true)
			con:Disconnect()
			con2:Disconnect()

			if weld then
				weld:Destroy()
			end
			
			victim:SetAttribute("carriedBy", "nill")
			attacker:SetAttribute("carrying", "nill")
			
			animator.StopAnimationOnHumanoid(attacker.Humanoid)
			
			warn(attacker.Name.. " dropped "..victim.Name)
			
			RagdollModule.Ragdoll(victim)
			if isVictimPlayer then
				RagdollForce:FireClient(game.Players:GetPlayerFromCharacter(victim))
			end
			module:iframes(victim, false)
			
			module:prompt_changer(true, true, true, false, false,victim, attacker)
			
			for _,v in victim:GetDescendants() do
				if v:IsA("BasePart") then
					v.Massless = false
				end
			end
			
			if attacker:FindFirstChild("carrying") then
				if attacker.carrying:IsA("BoolValue") then
					warn("destroyed carrying")
					attacker.carrying:Destroy()
				end
			end
			
			module:playSound(script:WaitForChild("Uncarry"), victimRoot)
			
			if dropPrompt then
				dropPrompt.Enabled = false
			end
			if finishHoldPrompt then
				finishHoldPrompt.Enabled = false
			end
			
			if isVictimPlayer then
				ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt}, false)
				ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(victim), true)
			end
			
			
			
			victim:SetAttribute("canGetUp", true)
			
			ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), true)
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), allWallPrompts, false)
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), {dropPrompt, finishHoldPrompt}, false)
			
			--animator.StopAnimation(victimAnim)
		elseif prompt.Name == "finishHoldPrompt" and prompt.Parent.Parent == victim and player.Character:GetAttribute("downed") == false then
			module:vic_collision(victim, true)
			--module:Clone_Bat(victim)
			con:Disconnect()
			con2:Disconnect()
			
			victim:SetAttribute("carriedBy", "nill")
			attacker:SetAttribute("carrying", "nill")
			
			if weld then
				weld:Destroy()
			end
			
			if attacker:FindFirstChild("carrying") then
				if attacker.carrying:IsA("BoolValue") then
					warn("destroyed carrying")
					attacker.carrying:Destroy()
				end
			end

			module:playSound(script:WaitForChild("Uncarry"), victimRoot)
			
			if dropPrompt then
				dropPrompt.Enabled = false
			end
			if finishHoldPrompt then
				finishHoldPrompt.Enabled = false
			end
			
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), {dropPrompt, finishHoldPrompt}, false)
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), allWallPrompts, false)
			
			
			module:grip(attacker, victim, "Finishing Hold")
		elseif prompt.Name == "Wall" and player.Character:GetAttribute("downed") == false then
			-- attackerRoot (Grab the carryWeld)
			if player.Character:GetAttribute("carrying") == victim.Name and victim:GetAttribute("carriedBy") == player.Name then
				module:vic_collision(victim, true)
				
				con:Disconnect()
				con2:Disconnect()

				if weld then
					weld:Destroy()
				end

				if attacker:FindFirstChild("carrying") then
					if attacker.carrying:IsA("BoolValue") then
						warn("destroyed carrying")
						attacker.carrying:Destroy()
					end
				end

				module:playSound(script:WaitForChild("Uncarry"), victimRoot)

				if dropPrompt then
					dropPrompt.Enabled = false
				end
				if finishHoldPrompt then
					finishHoldPrompt.Enabled = false
				end

				ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), {dropPrompt, finishHoldPrompt}, false)
				ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(attacker), allWallPrompts, false)

				--Wall Fix?
				victim:SetAttribute("carriedBy", "nill")
				attacker:SetAttribute("carrying", "nill")

				module:fun(attacker, victim, "Wall", prompt.Parent)
			end
		end
	end)
	
	con2 = attacker.ChildAdded:Connect(function(child)
		if child:IsA("BoolValue") and child.Name == "stun" then
			con:Disconnect()
			con2:Disconnect()
			
			if weld then
				weld:Destroy()
			end
			
			animator.StopAnimationOnHumanoid(attacker.Humanoid)
			
			warn(attacker.Name.. " was hit and dropped "..victim.Name)

			RagdollModule.Ragdoll(victim)
			RagdollForce:FireClient(game.Players:GetPlayerFromCharacter(victim))
			module:iframes(victim, false)

			module:prompt_changer(true, true, true, false, false,victim, attacker)

			for _,v in victim:GetDescendants() do
				if v:IsA("BasePart") then
					v.Massless = false
				end
			end
			
			if attacker:FindFirstChild("carrying") then
				if attacker.carrying:IsA("BoolValue") then
					warn("destroyed carrying")
					attacker.carrying:Destroy()
				end
			end
			
			module:playSound(script:WaitForChild("Uncarry"), victimRoot)
			ShowPrompt:FireClient(game.Players:GetPlayerFromCharacter(victim), {victim.Torso.executePrompt, victim.Torso.DefaultFun, victim.Torso.carryPrompt}, false)
			
			victim:SetAttribute("canGetUp", true)
			
			ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(attacker), true)
			ToggleResetting:FireClient(game.Players:GetPlayerFromCharacter(victim), true)
		end
		
	end)
end
